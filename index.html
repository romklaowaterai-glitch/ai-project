// โค้ดที่ถูกต้อง: รองรับทั้งการบันทึกข้อมูลจาก ESP32 และการดึงข้อมูลไปโชว์ที่หน้าเว็บ HTML function doGet(e) { return handleRequest(e); } function doPost(e) { return handleRequest(e); } function handleRequest(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("WaterData"); if (!sheet) { return ContentService.createTextOutput("ข้อผิดพลาด: ไม่พบแผ่นงานชื่อ WaterData กรุณาเปลี่ยนชื่อแผ่นงานด้านล่างให้ตรงกันก่อนครับ"); } // ➕ [เพิ่มจุดแก้ไขที่ 1] โหมดสำหรับหน้าเว็บ HTML มาดึงข้อมูลล่าสุดไปแสดงผล if (e.parameter.read === "latest") { var lastRow = sheet.getLastRow(); // ถ้าไม่มีข้อมูลในตารางเลย ให้ส่งค่าว่างกลับไป if (lastRow <= 1) { return ContentService.createTextOutput(JSON.stringify({ error: "No data found" })) .setMimeType(ContentService.MimeType.JSON); } // ดึงข้อมูลแถวสุดท้าย (ล่าสุด) มาจัดรูปแบบ var rowData = sheet.getRange(lastRow, 1, 1, 8).getValues()[0]; var result = { "time": rowData[0], "device": rowData[1], "temp": rowData[2], "ph": rowData[3], "turbidity": rowData[4], "tds": rowData[5], "waterColor": rowData[6], "accuracy": rowData[7] }; // ส่งข้อมูลกลับไปหาหน้าเว็บ HTML ในรูปแบบ JSON return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } // 📥 โหมดปกติ: รับข้อมูลจาก ESP32 เข้ามาบันทึกในตาราง var device = e.parameter.device || "ไม่ได้ระบุ"; var temp = e.parameter.temp || "-"; var ph = e.parameter.ph || "-"; var turbidity = e.parameter.turbidity || "-"; var tds = e.parameter.tds || "-"; var waterColor = e.parameter.waterColor || "-"; var accuracy = e.parameter.accuracy || "-"; // บันทึกลงตารางตามลำดับ A ถึง H sheet.appendRow([ new Date(), // A: วันที่-เวลา device, // B: อุปกรณ์ temp, // C: อุณหภูมิ ph, // D: ค่า pH turbidity, // E: ความขุ่น tds, // F: ค่า TDS waterColor, // G: สีของน้ำ accuracy // H: ความแม่นยำ ]); // เรียกใช้งานระบบส่งแจ้งเตือนเข้า Line Notify sendLineNotify(device, temp, ph, turbidity, tds, waterColor, accuracy); return ContentService.createTextOutput("บันทึกข้อมูลสำเร็จแล้ว! ลองเปิดดูใน Google Sheet ได้เลยครับ"); } function sendLineNotify(device, temp, ph, turbidity, tds, waterColor, accuracy) { var token = "ใส่_LINE_NOTIFY_TOKEN_ที่นี่"; // 👈 อย่าลืมนำ Token จริงมาใส่ตรงนี้นะคะ if(token.indexOf("ใส่_") !== -1 || token === "") return; // ข้ามถ้ายังไม่ได้ใส่ Token จริง var message = "\n📱 อุปกรณ์: " + device + "\n" + "🌡️ อุณหภูมิ: " + temp + " °C\n" + "🧪 ค่า pH: " + ph + "\n" + "☁️ ความขุ่น: " + turbidity + "\n" + "💧 ค่า TDS: " + tds + " ppm\n" + "🎨 สีของน้ำ: " + waterColor + "\n" + "🎯 ความแม่นยำ: " + accuracy; var options = { 'method': 'post', 'payload': {'message': message}, 'headers': {'Authorization': 'Bearer ' + token} }; try { // 🔧 [แก้ไขจุดที่ 2] เปลี่ยนลิงก์ปลายทางให้เป็น API จริงของ Line Notify UrlFetchApp.fetch('https://line.me', options); } catch(err) { Logger.log("การส่ง Line ล้มเหลว: " + err.toString()); } }
